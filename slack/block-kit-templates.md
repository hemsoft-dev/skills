# Slack Block Kit Message Templates

Complete, ready-to-use Block Kit templates for common messaging scenarios.

## Template 1: Deployment Notification

```powershell
$body = @{
    channel = "C08H7CG4NTS"
    text = "Deployment complete: api-gateway v2.3.1 to Production"
    blocks = @(
        @{
            type = "header"
            text = @{ type = "plain_text"; text = "🚀 Deployment Complete"; emoji = $true }
        }
        @{
            type = "section"
            fields = @(
                @{ type = "mrkdwn"; text = "*Service:*`napi-gateway" }
                @{ type = "mrkdwn"; text = "*Status:*`n✅ Success" }
                @{ type = "mrkdwn"; text = "*Environment:*`nProduction" }
                @{ type = "mrkdwn"; text = "*Version:*`nv2.3.1" }
            )
        }
        @{ type = "divider" }
        @{
            type = "context"
            elements = @(
                @{ type = "mrkdwn"; text = "Deployed by <@U2XMZDPJ7> • <!date^$([int](Get-Date -UFormat %s))^{date_short} at {time}|$(Get-Date -Format 'MMM d, yyyy h:mm tt')>" }
            )
        }
    )
} | ConvertTo-Json -Depth 10
```

## Template 2: Alert/Warning

```powershell
    text = "⚠️ High CPU usage detected on prod-web-01"

        @{

        @{
            type = "section"
            fields = @(
                @{ type = "mrkdwn"; text = "*Server:*`nprod-web-01" }
                @{ type = "mrkdwn"; text = "*CPU:*`n🔴 94%" }
                @{ type = "mrkdwn"; text = "*Memory:*`n🟡 72%" }
                @{ type = "mrkdwn"; text = "*Threshold:*`n80%" }

        }
        @{
            type = "actions"
            elements = @(
                @{
                    type = "button"
                    text = @{ type = "plain_text"; text = "📊 View Dashboard"; emoji = $true }
                    url = "https://monitoring.example.com"

                    type = "button"

            type = "context"

                @{ type = "mrkdwn"; text = "🤖 AlertBot • Triggered at $(Get-Date -Format 'h:mm tt')" }
            )
        }

```

## Template 3: Weekly Report

```powershell
$body = @{
    channel = "C08H7CG4NTS"
    text = "Weekly metrics report: Jan 1-7, 2026"
    blocks = @(
        @{
            type = "header"
            text = @{ type = "plain_text"; text = "📊 Weekly Performance Report"; emoji = $true }
        }
        @{
            type = "context"
            elements = @(
                @{ type = "mrkdwn"; text = "📅 January 1 - January 7, 2026" }
            )
        }
        @{
            type = "section"
            text = @{ type = "mrkdwn"; text = "*Highlights this week:*`n• 🟢 Uptime: *99.97%* (+0.02%)`n• 📈 Total requests: *1.2M* (+15%)`n• ⚡ Avg latency: *42ms* (-8ms)`n• 🐛 Error rate: *0.03%* (no change)" }
        }
        @{ type = "divider" }
        @{
            type = "section"
            fields = @(
                @{ type = "mrkdwn"; text = "*Top Endpoint:*`n`/api/users`" }
                @{ type = "mrkdwn"; text = "*Requests:*`n324,521" }
                @{ type = "mrkdwn"; text = "*Peak Hour:*`n2pm Tuesday" }
                @{ type = "mrkdwn"; text = "*Peak RPS:*`n1,247" }
            )
        }
        @{ type = "divider" }
        @{
            type = "context"
            elements = @(
                @{ type = "mrkdwn"; text = "🤖 Generated by MetricsBot • <https://dashboard.example.com|View full report>" }
            )
        }
    )
} | ConvertTo-Json -Depth 10
```

## Template 4: PR Review Request

```powershell
$body = @{
    channel = "C08H7CG4NTS"
    text = "New PR ready for review: #123 Add authentication flow"
    blocks = @(
        @{
            type = "header"
            text = @{ type = "plain_text"; text = "👀 Pull Request Ready for Review"; emoji = $true }
        }
        @{
            type = "section"
            text = @{ type = "mrkdwn"; text = "*<https://github.com/org/repo/pull/123|#123: Add OAuth2 authentication flow>*`n`nImplements Google and GitHub OAuth providers with JWT token refresh." }
        }
        @{
            type = "section"
            fields = @(
                @{ type = "mrkdwn"; text = "*Author:*`n<@U2XMZDPJ7>" }
                @{ type = "mrkdwn"; text = "*Status:*`n🟡 Pending Review" }
                @{ type = "mrkdwn"; text = "*Files:*`n12 changed" }
                @{ type = "mrkdwn"; text = "*Lines:*`n+234 / -56" }
            )
        }
        @{
            type = "actions"
            elements = @(
                @{
                    type = "button"
                    text = @{ type = "plain_text"; text = "👀 Review"; emoji = $true }
                    url = "https://github.com/org/repo/pull/123"
                }
                @{
                    type = "button"
                    text = @{ type = "plain_text"; text = "✅ Approve"; emoji = $true }
                    style = "primary"
                    value = "approve_123"
                }
            )
        }
        @{ type = "divider" }
        @{
            type = "context"
            elements = @(
                @{ type = "mrkdwn"; text = "🔔 Requested reviewers: <@U2XMZDPJ7>" }
            )
        }
    )
} | ConvertTo-Json -Depth 10
```
